const COLORS = { 'R':'#e74c3c', 'G':'#2ecc71', 'B':'#3498db', 'Y':'#f1c40f', 'P':'#ff66b2', 'O':'#ff8c00', 'C':'#1abc9c', 'T':'#9b59b6', 'M':'#d500f9' };

// ==========================================
//   ХЕЛПЕРЫ ГЕНЕРАЦИИ УРОВНЕЙ
// ==========================================
function hasAvailableMovesForBottles(bottles) {
    for (let i = 0; i < bottles.length; i++) {
        const src = bottles[i];
        if (src.length === 0) continue;
        const topColor = src[src.length - 1];
        for (let j = 0; j < bottles.length; j++) {
            if (i === j) continue;
            const dst = bottles[j];
            if (dst.length >= 4) continue;
            if (dst.length === 0 || dst[dst.length - 1] === topColor) {
                return true;
            }
        }
    }
    return false;
}

function generateLevel(totalBottles, hiddenCount, partialLengths, emptyCount, colorSet = COLOR_KEYS, maxAttempts = 100) {
    const fullBottles = totalBottles - emptyCount - partialLengths.length;
    let attempts = 0;
    while (attempts < maxAttempts) {
        attempts++;
        let bottles = [];
        for (let i = 0; i < fullBottles; i++) {
            let bottle = [];
            for (let j = 0; j < 4; j++) {
                bottle.push(colorSet[Math.floor(Math.random() * colorSet.length)]);
            }
            bottles.push(bottle);
        }
        for (let len of partialLengths) {
            let bottle = [];
            for (let j = 0; j < len; j++) {
                bottle.push(colorSet[Math.floor(Math.random() * colorSet.length)]);
            }
            bottles.push(bottle);
        }
        for (let i = 0; i < emptyCount; i++) {
            bottles.push([]);
        }
        let hiddenIndices = [];
        if (hiddenCount > 0) {
            let pool = Array.from({length: fullBottles}, (_,i) => i);
            for (let i = pool.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i+1));
                [pool[i], pool[j]] = [pool[j], pool[i]];
            }
            hiddenIndices = pool.slice(0, hiddenCount);
        }
        if (hasAvailableMovesForBottles(bottles)) {
            return { bottles, hiddenIndices };
        }
    }
    return { bottles: generateFallback(totalBottles, hiddenCount, partialLengths, emptyCount), hiddenIndices: [] };
}

function generateFallback(totalBottles, hiddenCount, partialLengths, emptyCount) {
    const fallback = [
        ['R','G','B','Y'], ['P','O','C','R'], ['G','B','Y','P'], ['O','C','R','G'],
        ['B','Y','P','O'], ['C','R','G','B'], ['Y','P','O','C'], []
    ];
    while (fallback.length < totalBottles) { fallback.push(['R','G','B','Y']); }
    return fallback.slice(0, totalBottles);
}

function makeDecorEl(d) {
    const el = document.createElement('div');
    el.className = 'pl-decor';
    el.style.left = d.x + '%';
    el.style.top = d.y + '%';
    el.style.width = d.w + '%';
    el.style.height = d.h + '%';
    el.style.background = d.c;
    if (d.t === 'crater') {
        el.style.borderRadius = '50%';
        el.style.boxShadow = 'inset 1.5px 1.5px 2px rgba(255,255,255,0.35), inset -1px -1px 2px rgba(0,0,0,0.25)';
    } else if (d.t === 'band') {
        el.style.borderRadius = '50%';
        el.style.transform = 'rotate(' + (d.r || 0) + 'deg)';
    } else {
        el.style.borderRadius = d.br || '50%';
        if (d.r) el.style.transform = 'rotate(' + d.r + 'deg)';
    }
    return el;
}

function makePlanetVisual(i, opts) {
    opts = opts || {};
    const tpl = PLANET_TPL[i % PLANET_TPL.length];
    const wrap = document.createElement('div');
    wrap.className = 'pl-wrap';

    if (!opts.mini) {
        const halo = document.createElement('div');
        halo.className = 'pl-halo';
        halo.style.background = 'radial-gradient(circle, ' + tpl.glow + ' 0%, transparent 68%)';
        wrap.appendChild(halo);
    }

    if (tpl.ring) {
        const rb = document.createElement('div');
        rb.className = 'pl-ring pl-ring-back';
        rb.style.borderColor = tpl.ring.c;
        rb.style.transform = 'rotate(' + tpl.ring.tilt + 'deg)';
        if (opts.mini) rb.style.borderWidth = '2px';
        wrap.appendChild(rb);
    }

    const body = document.createElement('div');
    body.className = 'pl-body';
    body.style.background = tpl.bg;
    tpl.decor.forEach(d => body.appendChild(makeDecorEl(d)));
    const shine = document.createElement('div');
    shine.className = 'pl-shine';
    body.appendChild(shine);
    wrap.appendChild(body);

    if (tpl.ring) {
        const rf = document.createElement('div');
        rf.className = 'pl-ring pl-ring-front';
        rf.style.borderColor = tpl.ring.c;
        rf.style.transform = 'rotate(' + tpl.ring.tilt + 'deg)';
        if (opts.mini) rf.style.borderWidth = '2px';
        wrap.appendChild(rf);
    }
    return wrap;
}
